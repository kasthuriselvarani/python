    @classmethod
    def Change_loc(cls):

    @staticmethod
    def Get_Password():
        pwd = int(input('enter 4 digit pin='))
        return pwd
    def Check_Bal(self):
        count = 3
        while count>0:
            print(f'Attempts left {count}')
            if self.Get_Password() == self.Pin:
                print(f'Available Bal is {self.Bal}')
                break
            else:
                print('Incorrect pin...')
                count -=1
        else:
            print('No attempts left')
            print('Try after 24 hours')
    def with_Drawn(self):
        amount = int(input('enter the amount ='))
        count =3
        while count>0:
            print(f'Attempts left {count}')
            if 100<=amount<=20000 and amount%100==0:

                    self.Bal -=amount

                    print(f'Available bal is {self.Bal}')
